class LoginPage extends StatefulWidget {
  final String language;

  const LoginPage({super.key, required this.language});

  @override
  State<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends State<LoginPage> {
  final emailController = TextEditingController();
  final passwordController = TextEditingController();

  String tr(String key) => T.get(widget.language, key);
  bool get isKo => widget.language == 'ko';

  void message(String msg) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(msg)));
  }

  void goWalletSetup() {
    Navigator.pushReplacement(
      context,
      MaterialPageRoute(
        builder: (_) => WalletSetupPage(language: widget.language),
      ),
    );
  }

  Future<void> createUserProfile(User user) async {
    final userDoc = FirebaseFirestore.instance
        .collection('users')
        .doc(user.uid);
    final snapshot = await userDoc.get();

    if (!snapshot.exists) {
      await userDoc.set({
        'email': user.email ?? '',
        'nickname': user.email?.split('@').first ?? 'ORQUBE Player',
        'country': 'Korea',
        'team': 'ORQUBE Alpha',
        'role': 'Rifleman',
        'bio': tr('earlyMember'),
        'points': 1250,
        'orbs': 125,
        'nftCount': 5,
        'selectedNft': AppAssets.logo,
        'attendanceCount': 12,
        'referralCount': 5,
        'createdAt': FieldValue.serverTimestamp(),
      });
    }
  }

  Future<void> emailSignUp() async {
    try {
      final result = await FirebaseAuth.instance.createUserWithEmailAndPassword(
        email: emailController.text.trim(),
        password: passwordController.text.trim(),
      );
      final user = result.user;
      if (user != null) await createUserProfile(user);
      goWalletSetup();
    } catch (e) {
      message('Sign-up error: $e');
    }
  }

  Future<void> emailLogin() async {
    try {
      final result = await FirebaseAuth.instance.signInWithEmailAndPassword(
        email: emailController.text.trim(),
        password: passwordController.text.trim(),
      );
      final user = result.user;
      if (user != null) await createUserProfile(user);
      goWalletSetup();
    } catch (e) {
      message('Login error: $e');
    }
  }

  Future<void> googleLogin() async {
    try {
      final googleUser = await GoogleSignIn().signIn();
      if (googleUser == null) return;

      final auth = await googleUser.authentication;
      final credential = GoogleAuthProvider.credential(
        accessToken: auth.accessToken,
        idToken: auth.idToken,
      );
      final result = await FirebaseAuth.instance.signInWithCredential(
        credential,
      );
      final firebaseUser = result.user;

      if (firebaseUser != null) await createUserProfile(firebaseUser);
      goWalletSetup();
    } catch (e) {
      message('Google login error: $e');
    }
  }

  void demoLogin() {
    Navigator.pushReplacement(
      context,
      MaterialPageRoute(
        builder:
            (_) => MiningDashboardV2(
              language: widget.language,
              walletStatus: tr('kaiaWalletPlanned'),
              walletAddress: '0xKaiaWallet...8f3d',
              profile: UserProfile(
                nickname: 'OrqubeUser',
                country: 'Korea',
                team: 'ORQUBE Alpha',
                role: 'Rifleman',
                bio: tr('earlyMember'),
                selectedNft: AppAssets.logo,
              ),
            ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final title = isKo ? 'ORQUBE 회원가입 / 로그인' : 'ORQUBE Sign Up / Login';

    return Scaffold(
      backgroundColor: AppColors.bg,

      body: Center(
        child: Container(
          width: 480,
          padding: const EdgeInsets.all(28),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Icon(Icons.security, color: AppColors.purple, size: 64),

              Text(
                title,
                style: const TextStyle(
                  fontSize: 28,
                  fontWeight: FontWeight.w900,
                ),
              ),
              const SizedBox(height: 20),
              TextField(
                controller: emailController,
                decoration: const InputDecoration(
                  labelText: 'Email',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 12),
              TextField(
                controller: passwordController,
                obscureText: true,
                decoration: const InputDecoration(
                  labelText: 'Password',
                  border: OutlineInputBorder(),
                ),
              ),
              const SizedBox(height: 12),
              Row(
                children: [
                  Expanded(
                    child: ElevatedButton(
                      onPressed: emailSignUp,

  });

  @override
  State<MiningDashboardV2> createState() => _MiningDashboardV2State();
}

class _MiningDashboardV2State extends State<MiningDashboardV2> {
  String tr(String key) => T.get(widget.language, key);
  int selectedIndex = 0;
  bool minedToday = false;
  bool isMining = false;

  Timer? timer;
  Duration remaining = Duration.zero;

  double minedOrbs = 0.0;
  final double dailyOrbs = 4.8;
  final int dailyPointReward = 20;
  final int adPointReward = 10;

  DateTime? miningStartTime;
  DateTime? miningEndTime;
  @override
  void initState() {
    super.initState();
    loadMiningState();

    timer = Timer.periodic(const Duration(seconds: 1), (_) {
      updateMiningCounter();
    });
  }

  @override
  void dispose() {
    timer?.cancel();
    super.dispose();
  }

  String todayKey() {
    final now = DateTime.now();
    return '${now.year}-${now.month.toString().padLeft(2, '0')}-${now.day.toString().padLeft(2, '0')}';
  }

  String get uid => FirebaseAuth.instance.currentUser?.uid ?? '';

  Future<void> loadMiningState() async {
    if (uid.isEmpty) {
      setState(() {
        minedToday = false;
        isMining = false;
        remaining = Duration.zero;
        minedOrbs = 0.0;
      });
      return;
    }

    final doc =
        await FirebaseFirestore.instance.collection('users').doc(uid).get();

    final data = doc.data();
    if (data == null) return;

    final Timestamp? startTs = data['miningStartTime'];
    final Timestamp? endTs = data['miningEndTime'];
    final String? lastKey = data['lastMiningDate'];
    final double savedMinedOrbs = (data['minedOrbs'] ?? 0.0).toDouble();

    final now = DateTime.now();

    miningStartTime = startTs?.toDate();
    miningEndTime = endTs?.toDate();

    if (miningStartTime != null &&
        miningEndTime != null &&
        now.isBefore(miningEndTime!)) {
      final elapsedSeconds = now.difference(miningStartTime!).inSeconds;
      final totalSeconds = 24 * 60 * 60;
      final calculatedOrbs = dailyOrbs * (elapsedSeconds / totalSeconds);

      setState(() {
        minedToday = lastKey == todayKey();
        isMining = true;
        remaining = miningEndTime!.difference(now);
        minedOrbs = calculatedOrbs.clamp(0.0, dailyOrbs);
      });
    } else {
      setState(() {
        minedToday = false;
        isMining = false;
        remaining = Duration.zero;
        minedOrbs = savedMinedOrbs;
      });
    }
  }

  void updateMiningCounter() {
    if (!mounted) return;

    if (miningStartTime == null || miningEndTime == null) return;

    final now = DateTime.now();

    if (now.isAfter(miningEndTime!)) {
      setState(() {
        isMining = false;
        remaining = Duration.zero;
        minedOrbs = dailyOrbs;
        minedToday = false;
      });
      return;
    }

    final elapsedSeconds = now.difference(miningStartTime!).inSeconds;
    final totalSeconds = 24 * 60 * 60;
    final calculatedOrbs = dailyOrbs * (elapsedSeconds / totalSeconds);

    setState(() {
      isMining = true;
      remaining = miningEndTime!.difference(now);
      minedOrbs = calculatedOrbs.clamp(0.0, dailyOrbs);
    });
  }

  Future<void> saveMiningState() async {
    if (uid.isEmpty) return;

    await FirebaseFirestore.instance.collection('users').doc(uid).set({
      'points': widget.profile.points,
      'orbs': widget.profile.orbs,
      'attendanceCount': widget.profile.attendanceCount,
      'minedOrbs': minedOrbs,
      'lastMiningDate': todayKey(),
      'miningStartTime': miningStartTime,
      'miningEndTime': miningEndTime,
      'updatedAt': FieldValue.serverTimestamp(),
    }, SetOptions(merge: true));
  }

  String get pointText =>
      '${widget.profile.points.toString().replaceAllMapped(RegExp(r'\B(?=(\d{3})+(?!\d))'), (match) => ',')}P';

  String durationText(Duration d) {
    final h = d.inHours.toString().padLeft(2, '0');
    final m = (d.inMinutes % 60).toString().padLeft(2, '0');
    final s = (d.inSeconds % 60).toString().padLeft(2, '0');
    return '$h:$m:$s';
  }

  void showMessage(String msg) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(msg)));
  }

  void dailyMining() async {
    if (isMining && remaining.inSeconds > 0) {
      showMessage('아직 채굴 중입니다. 남은 시간: ${durationText(remaining)}');
      return;
    }

    final now = DateTime.now();

    setState(() {
      widget.profile.points += dailyPointReward;
      widget.profile.orbs = widget.profile.points ~/ 10;
      widget.profile.attendanceCount += 1;

      minedToday = true;
      isMining = true;
      minedOrbs = 0.0;

      miningStartTime = now;
      miningEndTime = now.add(const Duration(hours: 24));
      remaining = const Duration(hours: 24);
    });

    await saveMiningState();

    showMessage('오늘의 채굴 시작! +$dailyPointReward P 지급되었습니다.');
  }

  void addBonus(String kind, int amount) {
    setState(() {
      widget.profile.points += amount;
      widget.profile.orbs = widget.profile.points ~/ 10;
      if (kind == 'checkin') widget.profile.attendanceCount++;
      if (kind == 'invite') widget.profile.referralCount++;
    });
    showMessage('+$amount P ${tr('bonusGiven')}');
  }

  Future<void> watchAdReward() async {
    await AdMobHelper.showRewardedAd(
      onRewarded: () {
        setState(() {
          widget.profile.points += adPointReward;
          widget.profile.orbs = widget.profile.points ~/ 10;
        });

      onUnavailable: () {
        showMessage('광고를 불러오는 중입니다. 잠시 후 다시 눌러주세요.');
      },
    );
  }

class MiningRingPainter extends CustomPainter {
  final double progress;
  MiningRingPainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final center = size.center(Offset.zero);
    final rect = Rect.fromCircle(center: center, radius: size.width / 2 - 16);

    final base =
        Paint()
          ..color = Colors.white.withOpacity(0.07)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 14
          ..strokeCap = StrokeCap.round;

    final ring =
        Paint()
          ..shader = const SweepGradient(
            colors: [
              AppColors.violet,
              Color(0xFFE9D5FF),
              AppColors.violet2,
              AppColors.violet,
            ],
          ).createShader(rect)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 14
          ..strokeCap = StrokeCap.round;

    canvas.drawArc(rect, math.pi * 0.80, math.pi * 1.45, false, base);
    canvas.drawArc(
      rect,
      math.pi * 0.80,
      math.pi * 1.45 * progress,
      false,
      ring,
    );

    final ring2 =
        Paint()
          ..color = AppColors.violet.withOpacity(0.55)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 5
          ..strokeCap = StrokeCap.round;
    canvas.drawArc(
      Rect.fromCircle(center: center, radius: size.width / 2 - 42),
      math.pi * 1.75,
      math.pi * 0.35,
      false,
      ring2,
    );
  }

  @override
  bool shouldRepaint(covariant MiningRingPainter oldDelegate) =>
      oldDelegate.progress != progress;
}

class BoostRingPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final rect = Rect.fromCircle(
      center: size.center(Offset.zero),
      radius: size.width / 2 - 10,
    );
    final base =
        Paint()
          ..color = Colors.white.withOpacity(0.08)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 13;

    final ring =
        Paint()
          ..shader = const SweepGradient(
            colors: [
              AppColors.violet,
              Color(0xFFE9D5FF),
              AppColors.violet2,
              AppColors.violet,
            ],
          ).createShader(rect)
          ..style = PaintingStyle.stroke
          ..strokeWidth = 13
          ..strokeCap = StrokeCap.round;

    canvas.drawCircle(size.center(Offset.zero), size.width / 2 - 10, base);
    canvas.drawArc(rect, -math.pi / 2, math.pi * 2 * 0.90, false, ring);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}
