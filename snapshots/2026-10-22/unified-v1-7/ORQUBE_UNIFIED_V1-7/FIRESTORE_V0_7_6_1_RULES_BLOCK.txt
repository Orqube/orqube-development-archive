    match /launch_projects/{projectId} {
      // ORQUBE AI Launch rule
      // ORQUBE AI Launch rule
      allow read: if
        resource.data.status == 'published'
        || isAdmin()
        || (signedIn() && resource.data.ownerUid == request.auth.uid);

      // ORQUBE AI Launch rule
      allow create: if signedIn()
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.status == 'draft'
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 80
        && request.resource.data.symbol is string
        && request.resource.data.symbol.size() >= 1
        && request.resource.data.symbol.size() <= 16
        && request.resource.data.description is string
        && request.resource.data.description.size() <= 3000
        && request.resource.data.chain == 'Kaia'
        && request.resource.data.contractAddress == null;

      // ORQUBE AI Launch rule
      // ORQUBE AI Launch rule
      allow update: if isAdmin() || (
        signedIn()
        && resource.data.ownerUid == request.auth.uid
        && request.resource.data.ownerUid == resource.data.ownerUid
        && !request.resource.data.diff(resource.data).affectedKeys().hasAny([
          'contractAddress',
          'deployTxHash',
          'txHash',
          'deployedAt',
          'chainId',
          'price',
          'priceChange24h',
          'holders',
          'volume24h',
          'marketCap',
          'externalMarketUrl'
        ])
        && (
          // ORQUBE AI Launch rule
          !request.resource.data.diff(resource.data).affectedKeys().hasAny([
            'deploymentStatus',
            'deploymentRequestId'
          ])

          // ORQUBE AI Launch rule
          || (
            request.resource.data.diff(resource.data).affectedKeys().hasOnly([
              'walletAddress',
              'deploymentStatus',
              'updatedAt'
            ])
            && request.resource.data.deploymentStatus == 'wallet_ready'
            && request.resource.data.walletAddress is string
            && request.resource.data.walletAddress.matches('^0x[0-9a-fA-F]{40}$')
          )

          // ORQUBE AI Launch rule
          || (
            request.resource.data.diff(resource.data).affectedKeys().hasOnly([
              'walletAddress',
              'deploymentStatus',
              'deploymentRequestId',
              'updatedAt'
            ])
            && request.resource.data.deploymentStatus == 'review_requested'
            && request.resource.data.walletAddress is string
            && request.resource.data.walletAddress.matches('^0x[0-9a-fA-F]{40}$')
            && request.resource.data.deploymentRequestId is string
            && existsAfter(
              /databases/$(database)/documents/launch_deployment_requests/$(request.resource.data.deploymentRequestId)
            )
            && getAfter(
              /databases/$(database)/documents/launch_deployment_requests/$(request.resource.data.deploymentRequestId)
            ).data.projectId == projectId
            && getAfter(
              /databases/$(database)/documents/launch_deployment_requests/$(request.resource.data.deploymentRequestId)
            ).data.ownerUid == request.auth.uid
          )
        )
      );

      allow delete: if isAdmin()
        || (signedIn() && resource.data.ownerUid == request.auth.uid);
    }

    match /launch_wallets/{ownerUid} {
      // ORQUBE AI Launch rule
      allow read: if owner(ownerUid) || isAdmin();

      // ORQUBE AI Launch rule
      // ORQUBE AI Launch rule
      allow create: if owner(ownerUid)
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.walletAddress is string
        && request.resource.data.walletAddress.matches('^0x[0-9a-fA-F]{40}$')
        && request.resource.data.chain == 'Kaia'
        && request.resource.data.walletStatus == 'connected'
        && request.resource.data.keys().hasOnly([
          'ownerUid',
          'walletAddress',
          'chain',
          'walletStatus',
          'createdAt',
          'updatedAt'
        ]);

      allow update: if isAdmin() || (
        owner(ownerUid)
        && request.resource.data.ownerUid == resource.data.ownerUid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'ownerUid',
          'walletAddress',
          'chain',
          'walletStatus',
          'createdAt',
          'updatedAt'
        ])
        && (
          (
            request.resource.data.walletStatus == 'connected'
            && request.resource.data.walletAddress is string
            && request.resource.data.walletAddress.matches('^0x[0-9a-fA-F]{40}$')
            && request.resource.data.chain == 'Kaia'
          )
          || (
            request.resource.data.walletStatus == 'disconnected'
            && request.resource.data.walletAddress == null
          )
        )
      );

      allow delete: if isAdmin();
    }

    match /launch_starter_requests/{ownerUid} {
      // ORQUBE AI Launch rule
      allow read: if owner(ownerUid) || isAdmin();

      // ORQUBE AI Launch rule
      allow create: if owner(ownerUid)
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.keys().hasOnly([
          'ownerUid',
          'status',
          'requestedAt',
          'updatedAt'
        ]);

      // ORQUBE AI Launch rule
      // ORQUBE AI Launch rule
      allow update: if isAdmin() || (
        owner(ownerUid)
        && resource.data.ownerUid == request.auth.uid
        && request.resource.data.ownerUid == resource.data.ownerUid
        && resource.data.status in ['not_requested', 'rejected']
        && request.resource.data.status == 'pending'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'status',
          'requestedAt',
          'updatedAt'
        ])
      );

      allow delete: if isAdmin();
    }

        && request.resource.data.projectId is string

        && request.resource.data.deployFeeOrbs is number
        && request.resource.data.deployFeeOrbs > 0
        && request.resource.data.status == 'review_requested'
        && request.resource.data.keys().hasOnly([
          'projectId',
          'ownerUid',
          'walletAddress',
          'deployFeeOrbs',
          'status',
          'createdAt'
        ])
        && exists(
          /databases/$(database)/documents/launch_projects/$(request.resource.data.projectId)
        )
        && get(
          /databases/$(database)/documents/launch_projects/$(request.resource.data.projectId)
        ).data.ownerUid == request.auth.uid
        && get(
          /databases/$(database)/documents/launch_projects/$(request.resource.data.projectId)
        ).data.contractAddress == null
        && exists(
          /databases/$(database)/documents/launch_config/token_deploy
        )
        && request.resource.data.deployFeeOrbs
          == get(
            /databases/$(database)/documents/launch_config/token_deploy
          ).data.deployFeeOrbs;

      allow update, delete: if isAdmin();
