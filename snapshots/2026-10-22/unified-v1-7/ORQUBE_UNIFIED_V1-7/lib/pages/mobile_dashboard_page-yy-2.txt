  String _dayKey([DateTime? value]) {
    final d = (value ?? DateTime.now()).toLocal();
    return '${d.year.toString().padLeft(4, '0')}-'
        '${d.month.toString().padLeft(2, '0')}-'
        '${d.day.toString().padLeft(2, '0')}';
  }

  int _isoWeekNumber(DateTime date) {
    final d = DateTime(date.year, date.month, date.day);
    final thursday = d.add(Duration(days: 4 - d.weekday));
    final firstThursday = DateTime(thursday.year, 1, 4);
    final firstWeekThursday = firstThursday.add(
      Duration(days: 4 - firstThursday.weekday),
    );
    return 1 + thursday.difference(firstWeekThursday).inDays ~/ 7;
  }

  String _weekKey([DateTime? value]) {
    final d = (value ?? DateTime.now()).toLocal();
    final thursday = d.add(Duration(days: 4 - d.weekday));
    final weekYear = thursday.year;
    final week = _isoWeekNumber(d);
    return '$weekYear-W${week.toString().padLeft(2, '0')}';
  }

    if (mounted) {

  Future<void> _loadCryptoMarket({bool showFeedback = false}) async {
    if (_cryptoLoading) return;

    if (mounted) {
      setState(() {
        _cryptoLoading = true;
        if (_cryptoQuotes.isEmpty) _cryptoError = null;
      });
    } else {
      _cryptoLoading = true;
    }

    try {
      final response = await http
          .get(Uri.parse(_cryptoMarketEndpoint))
          .timeout(const Duration(seconds: 10));

      if (response.statusCode != 200) {
        throw Exception('HTTP ${response.statusCode}');
      }

      final decoded = jsonDecode(response.body);
      if (decoded is! List) {
        throw const FormatException('Unexpected market response');
      }

      final quotes = <_CryptoQuote>[];
      for (final raw in decoded) {
        if (raw is! Map<String, dynamic>) continue;
        final quote = _CryptoQuote.fromJson(raw);
        if (quote.price > 0) quotes.add(quote);
      }

      if (quotes.isEmpty) {
        throw const FormatException('No market quotes received');
      }

      // 화면 순서는 ORQUBE에서 보기 좋은 고정 순서를 유지합니다.
      const order = <String>['BTC', 'ETH', 'BNB', 'SOL', 'XRP', 'KAIA'];
      quotes.sort((a, b) {
        final ai = order.indexOf(a.symbol);
        final bi = order.indexOf(b.symbol);
        return (ai < 0 ? 999 : ai).compareTo(bi < 0 ? 999 : bi);
      });

      if (!mounted) return;
      setState(() {
        _cryptoQuotes
          ..clear()
          ..addAll(quotes);
        _cryptoUpdatedAt = DateTime.now();
        _cryptoError = null;
      });

      if (showFeedback) {
        _message(
          _tr(
            'GLOBAL CRYPTO MARKET 시세를 새로고침했습니다.',
            'GLOBAL CRYPTO MARKET refreshed.',
            ja: 'GLOBAL CRYPTO MARKETを更新しました。',
            zh: 'GLOBAL CRYPTO MARKET 已刷新。',
            fr: 'GLOBAL CRYPTO MARKET a été actualisé.',
            pt: 'GLOBAL CRYPTO MARKET foi atualizado.',
          ),
        );
      }
    } catch (e) {
      if (!mounted) return;
      setState(() => _cryptoError = e.toString());
      if (showFeedback) {
        _message(
          _tr(
            '실시간 시세를 불러오지 못했습니다. 잠시 후 다시 시도해주세요.',
            'Could not load live prices. Please try again shortly.',
            ja: 'リアルタイム価格を取得できませんでした。しばらくしてから再試行してください。',
            zh: '无法加载实时价格，请稍后重试。',
            fr: 'Impossible de charger les cours en direct. Réessayez bientôt.',
            pt: 'Não foi possível carregar as cotações ao vivo. Tente novamente em breve.',
          ),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _cryptoLoading = false);
      } else {
        _cryptoLoading = false;
      }
    }
  }

  String _formatUsd(double value) {
    final decimals = value >= 1000
        ? 2
        : value >= 1
        ? 3
        : value >= 0.01
        ? 4
        : 6;
    final raw = value.toStringAsFixed(decimals);
    final parts = raw.split('.');
    final whole = parts.first;
    final buffer = StringBuffer();
    for (int i = 0; i < whole.length; i++) {
      final remaining = whole.length - i;
      buffer.write(whole[i]);
      if (remaining > 1 && remaining % 3 == 1) buffer.write(',');
    }
    return '\$${buffer.toString()}${parts.length > 1 ? '.${parts[1]}' : ''}';
  }

  Future<void> _loadUserData() async {
    final docRef = _userDoc;
    if (docRef == null) {
      _restoreLocalDefaults();
      return;
    }

    try {
      final snapshot = await docRef.get();
      final data = snapshot.data();

      if (data == null) {
        await docRef.set({
          'email': _user?.email ?? '',
          'nickname': _nickname,
          'level': _level,
          'rank': _rank,
          'points': _points,
          'orbs': _storedOrbs,
          'attendance': _attendance,
          'referralCount': _referral,
          'nftCount': _nftCount,
          'walletAddress': _walletAddress,
          'language': _languageCode,
          'miningActive': false,
          'miningRate': 1.0,
          'miningRateBonuses': <String, dynamic>{},
          'dailyDogTag': null,
          'dailyDogTagDayKey': null,
          'dogTagCount': 0,
          'weeklyCodeEventId': null,
          'weeklyOperationWins': 0,
          'createdAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        }, SetOptions(merge: true));
        _restoreLocalDefaults();
        return;
      }

      _nickname =
          (data['nickname'] as String?)?.trim().isNotEmpty == true
              ? data['nickname'] as String
              : _nickname;
      _level = data['level']?.toString() ?? _level;
      _rank = data['rank']?.toString() ?? _rank;
      _points = _readInt(data['points'], _points);
      _storedOrbs = _readDouble(data['orbs'], _storedOrbs);
      _miningRate = math.max(1.0, _readDouble(data['miningRate'], 1.0));
      _attendance = _readInt(data['attendance'], _attendance);
      _referral = _readInt(data['referralCount'], _referral);
      _nftCount = _readInt(data['nftCount'], _nftCount);
      _walletAddress =
          data['walletAddress']?.toString().trim().isNotEmpty == true
              ? data['walletAddress'].toString()
              : _walletAddress;
      final savedLanguage = data['language']?.toString();
      if (savedLanguage != null && _languageLabels.containsKey(savedLanguage)) {
        _languageCode = savedLanguage;
        OrqubeI18n.setLanguage(savedLanguage);
      }
      _dailyClaimed = _isSameLocalDate(
        _timestampToDate(data['lastDailyRewardAt']),
        DateTime.now(),
      );

      _todayDogTag = data['dailyDogTag']?.toString();
      _lastDogTagDayKey = data['dailyDogTagDayKey']?.toString();
      _dogTagCount = _readInt(data['dogTagCount'], 0);
      _dogTagClaimedToday = _lastDogTagDayKey == _dayKey();
      _completedWeeklyEventId = data['weeklyCodeEventId']?.toString();
      _weeklyOperationWins = _readInt(data['weeklyOperationWins'], 0);
      _weeklyCodeCompleted = _completedWeeklyEventId == _weekKey();

      final startedAt = _timestampToDate(data['miningStartedAt']);
      final endsAt = _timestampToDate(data['miningEndsAt']);
      final savedActive = data['miningActive'] == true;

      _miningStartedAt = startedAt;
      _miningEndsAt = endsAt;
      _isMining =
          savedActive &&
          startedAt != null &&
          endsAt != null &&
          DateTime.now().isBefore(endsAt);

      if (savedActive && !_isMining) {
        await _finishMiningSession(persist: true);
      } else {
        _recalculateDisplayOrbs();
      }
    } catch (e) {
      _restoreLocalDefaults();
      _message('저장된 채굴 정보를 불러오지 못했습니다: $e');
    }

    if (mounted) setState(() {});
  }

  void _restoreLocalDefaults() {
    _isMining = false;
    _miningStartedAt = null;
    _miningEndsAt = null;
    _displayOrbs = _storedOrbs;
  }

  int _readInt(dynamic value, int fallback) {
    if (value is int) return value;
    if (value is num) return value.toInt();
    return int.tryParse(value?.toString() ?? '') ?? fallback;
  }

  double _readDouble(dynamic value, double fallback) {
    if (value is num) return value.toDouble();
    return double.tryParse(value?.toString() ?? '') ?? fallback;
  }

  DateTime? _timestampToDate(dynamic value) {
    if (value is Timestamp) return value.toDate();
    if (value is DateTime) return value;
    return null;
  }

  bool _isSameLocalDate(DateTime? a, DateTime b) {
    if (a == null) return false;
    final localA = a.toLocal();
    final localB = b.toLocal();
    return localA.year == localB.year &&
        localA.month == localB.month &&
        localA.day == localB.day;
  }

  void _onTick() {
    if (!_isMining) return;

    final endsAt = _miningEndsAt;
    if (endsAt == null || !DateTime.now().isBefore(endsAt)) {
      _finishMiningSession(persist: true);
      return;
    }

    _recalculateDisplayOrbs();

    final now = DateTime.now();
    if (_lastPersistAt == null ||
        now.difference(_lastPersistAt!) >= const Duration(seconds: 30)) {
      _lastPersistAt = now;
      _persistMiningState();
    }

    if (mounted) setState(() {});
  }

  void _recalculateDisplayOrbs() {
    if (!_isMining || _miningStartedAt == null || _miningEndsAt == null) {
      _displayOrbs = _storedOrbs;
      return;
    }

    final now = DateTime.now();
    final effectiveNow = now.isAfter(_miningEndsAt!) ? _miningEndsAt! : now;
    final elapsed =
        effectiveNow.difference(_miningStartedAt!).inMilliseconds / 1000.0;
    _displayOrbs =
        _storedOrbs + math.max(0, elapsed) * orbsPerSecond * _miningRate;
  }

  Duration get _remaining {
    if (!_isMining || _miningEndsAt == null) return Duration.zero;
    final duration = _miningEndsAt!.difference(DateTime.now());
    return duration.isNegative ? Duration.zero : duration;
  }

  Duration get _elapsed {
    if (!_isMining || _miningStartedAt == null) return Duration.zero;
    final end =
        DateTime.now().isAfter(_miningEndsAt ?? DateTime.now())
            ? (_miningEndsAt ?? DateTime.now())
            : DateTime.now();
    final duration = end.difference(_miningStartedAt!);
    return duration.isNegative ? Duration.zero : duration;
  }

  String _durationText(Duration value) {
    final h = value.inHours.toString().padLeft(2, '0');
    final m = (value.inMinutes % 60).toString().padLeft(2, '0');
    final s = (value.inSeconds % 60).toString().padLeft(2, '0');
    return '$h:$m:$s';
  }

  String _remainingText(Duration value) {
    final h = value.inHours;
    final m = value.inMinutes % 60;
    final s = value.inSeconds % 60;
    return '${h}H ${m.toString().padLeft(2, '0')}M '
        '${s.toString().padLeft(2, '0')}S';
  }

  Future<void> _startMiningRequest() async {
    if (_startingMining || _isMining) return;

    setState(() => _startingMining = true);

    if (kIsWeb) {
      _message('Chrome에서는 AdMob이 지원되지 않아 테스트 채굴을 바로 시작합니다.');
      await _beginMiningSession();
      if (mounted) setState(() => _startingMining = false);
      return;
    }

    if (_rewardedAd == null) {
      _message('광고를 준비하고 있습니다. 잠시 후 다시 눌러주세요.');
      _loadRewardedAd();
      if (mounted) setState(() => _startingMining = false);
      return;
    }

    final ad = _rewardedAd;
    _rewardedAd = null;

    bool rewardEarned = false;

    ad!.fullScreenContentCallback = FullScreenContentCallback(
      onAdDismissedFullScreenContent: (ad) async {
        ad.dispose();
        _loadRewardedAd();

        if (rewardEarned) {
          await _applyMiningRateBonus(
            source: 'ad_start',
            amount: 0.05,
            description: '광고 시청 후 채굴 시작',
          );
          await _beginMiningSession();
        } else {
          _message('광고를 끝까지 시청해야 채굴이 시작됩니다.');
        }

        if (mounted) setState(() => _startingMining = false);
      },
      onAdFailedToShowFullScreenContent: (ad, error) {
        ad.dispose();
        _loadRewardedAd();
        _message('광고를 표시하지 못했습니다. 다시 시도해주세요.');
        if (mounted) setState(() => _startingMining = false);
      },
    );

    ad.show(
      onUserEarnedReward: (adWithoutView, rewardItem) {
        rewardEarned = true;
      },
    );
  }

  void _loadRewardedAd() {
    if (kIsWeb || _adLoading || _rewardedAd != null) return;

    String? adUnitId;
    if (!_adminAdEnabled) return;

    switch (defaultTargetPlatform) {
      case TargetPlatform.android:
        final productionId = _adminAndroidRewardedId.isNotEmpty
            ? _adminAndroidRewardedId
            : _androidRewardedProductionId;
        adUnitId = _adminUseTestAds || productionId.isEmpty
            ? _androidRewardedTestId
            : productionId;
        break;
      case TargetPlatform.iOS:
        final productionId = _adminIosRewardedId.isNotEmpty
            ? _adminIosRewardedId
            : _iosRewardedProductionId;
        adUnitId = _adminUseTestAds || productionId.isEmpty
            ? _iosRewardedTestId
            : productionId;
        break;
      default:
        adUnitId = null;
        break;
    }

    if (adUnitId == null) return;

    _adLoading = true;

    RewardedAd.load(
      adUnitId: adUnitId,
      request: const AdRequest(),
      rewardedAdLoadCallback: RewardedAdLoadCallback(
        onAdLoaded: (ad) {
          _rewardedAd?.dispose();
          _rewardedAd = ad;
          _adLoading = false;
          if (mounted) setState(() {});
        },
        onAdFailedToLoad: (error) {
          _rewardedAd = null;
          _adLoading = false;
          debugPrint('AdMob rewarded load failed: ${error.code} ${error.message}');
          if (mounted) setState(() {});
        },
      ),
    );
  }

  Future<void> _beginMiningSession() async {
    final now = DateTime.now();
    final endsAt = now.add(miningDuration);

    setState(() {
      _isMining = true;
      _miningStartedAt = now;
      _miningEndsAt = endsAt;
      _displayOrbs = _storedOrbs;
      _lastPersistAt = now;
    });

    final docRef = _userDoc;
    if (docRef != null) {
      try {
        await docRef.set({
          'miningActive': true,
          'miningStartedAt': Timestamp.fromDate(now),
          'miningEndsAt': Timestamp.fromDate(endsAt),
          'orbs': _storedOrbs,
          'points': _points,
          'updatedAt': FieldValue.serverTimestamp(),
        }, SetOptions(merge: true));
      } catch (e) {
        _message('채굴 시작 정보 저장에 실패했습니다: $e');
      }
    }

    _message('광고 시청이 확인되어 24시간 채굴을 시작했습니다.');
  }

  Future<void> _finishMiningSession({required bool persist}) async {
    if (!_isMining && _miningStartedAt == null) return;

    final startedAt = _miningStartedAt;
    final endsAt = _miningEndsAt;
    final effectiveEnd =
        endsAt != null && DateTime.now().isAfter(endsAt)
            ? endsAt
            : DateTime.now();

    if (startedAt != null) {
      final elapsed =
          effectiveEnd.difference(startedAt).inMilliseconds / 1000.0;
      _storedOrbs += math.max(0, elapsed) * orbsPerSecond;
    }

    _displayOrbs = _storedOrbs;
    _isMining = false;
    _miningStartedAt = null;
    _miningEndsAt = null;

    if (mounted) setState(() {});

    if (persist) {
      final docRef = _userDoc;
      if (docRef != null) {
        try {
          await docRef.set({
            'orbs': _storedOrbs,
            'points': _points,
            'miningActive': false,
          'miningRate': 1.0,
          'miningRateBonuses': <String, dynamic>{},
            'miningStartedAt': FieldValue.delete(),
            'miningEndsAt': FieldValue.delete(),
            'lastMiningCompletedAt': FieldValue.serverTimestamp(),
            'updatedAt': FieldValue.serverTimestamp(),
          }, SetOptions(merge: true));
        } catch (_) {}
      }
    }

    _message('24시간 채굴이 종료되었습니다. 광고 시청 후 다시 시작해주세요.');
  }

  Future<void> _persistMiningState() async {
    final docRef = _userDoc;
    if (docRef == null) return;

    _recalculateDisplayOrbs();

    try {
      await docRef.set({
        'orbs': _storedOrbs,
        'points': _points,
        'miningActive': _isMining,
        if (_isMining && _miningStartedAt != null)
          'miningStartedAt': Timestamp.fromDate(_miningStartedAt!),
        if (_isMining && _miningEndsAt != null)
          'miningEndsAt': Timestamp.fromDate(_miningEndsAt!),
        'updatedAt': FieldValue.serverTimestamp(),
      }, SetOptions(merge: true));
    } catch (_) {}
  }


  Future<void> _applyMiningRateBonus({
    required String source,
    required double amount,
    required String description,
  }) async {
    final user = _user;
    final docRef = _userDoc;
    if (user == null || docRef == null || amount <= 0) return;

    final eventRef = docRef.collection('miningRateHistory').doc();
    try {
      await FirebaseFirestore.instance.runTransaction((tx) async {
        final snap = await tx.get(docRef);
        final data = snap.data() ?? <String, dynamic>{};
        final current = math.max(1.0, _readDouble(data['miningRate'], 1.0));
        final next = double.parse((current + amount).toStringAsFixed(2));
        tx.set(docRef, {
          'miningRate': next,
          'miningRateUpdatedAt': FieldValue.serverTimestamp(),
          'updatedAt': FieldValue.serverTimestamp(),
        }, SetOptions(merge: true));
        tx.set(eventRef, {
          'uid': user.uid,
          'source': source,
          'amount': amount,
          'rateBefore': current,
          'rateAfter': next,
          'description': description,
          'createdAt': FieldValue.serverTimestamp(),
        });
        _miningRate = next;
      });
      if (mounted) setState(() {});
    } catch (e) {
      _message('채굴률 보너스 저장에 실패했습니다: $e');
    }
  }

  String _generateDogTag() {
    const units = <String>[
      'ALPHA',
      'BRAVO',
      'CHARLIE',
      'DELTA',
      'ECHO',
      'FALCON',
      'ORBIT',
      'VECTOR',
    ];
    final random = math.Random.secure();
    final unit = units[random.nextInt(units.length)];
    final number = 1 + random.nextInt(999);
    return '$unit-${number.toString().padLeft(3, '0')}';
  }

      _showDogTagDialog(_todayDogTag ?? '지급 완료');
      return;

    final newTag = _generateDogTag();
    try {
      await FirebaseFirestore.instance.runTransaction((transaction) async {
        final snapshot = await transaction.get(docRef);
        final data = snapshot.data() ?? <String, dynamic>{};
        final savedDayKey = data['dailyDogTagDayKey']?.toString();
        if (savedDayKey == todayKey) {
          throw const _AlreadyClaimedException();

        final currentPoints = _readInt(data['points'], _points);
        final currentOrbs = _readDouble(data['orbs'], _storedOrbs);
        final currentCount = _readInt(data['dogTagCount'], _dogTagCount);

        transaction.set(docRef, {
          'dailyDogTag': newTag,
          'dailyDogTagDayKey': todayKey,
          'dailyDogTagClaimedAt': FieldValue.serverTimestamp(),
          'dogTagCount': currentCount + 1,
          'points': currentPoints + 30,
          'orbs': currentOrbs + 0.0300,
          'updatedAt': FieldValue.serverTimestamp(),
        }, SetOptions(merge: true));
      });

      if (!mounted) return;
      setState(() {
        _todayDogTag = newTag;
        _lastDogTagDayKey = todayKey;
        _dogTagClaimedToday = true;
        _dogTagCount += 1;
        _points += 30;
        _storedOrbs += 0.0300;
        _displayOrbs += 0.0300;
      });
      _showDogTagDialog(newTag);
    } on _AlreadyClaimedException {
      await _loadUserData();
      _message('오늘의 군번줄은 이미 받았습니다.');
    } catch (e) {

  Future<void> _completeWeeklyCode(_WeeklyOperation operation) async {

      _message('로그인 후 주간 작전에 참여할 수 있습니다.');
      return;

    final eventId = _weekKey();
    try {
      await FirebaseFirestore.instance.runTransaction((transaction) async {

        final data = snapshot.data() ?? <String, dynamic>{};
        if (data['weeklyCodeEventId']?.toString() == eventId) {
          throw const _AlreadyClaimedException();
        }

        final currentPoints = _readInt(data['points'], _points);
        final currentOrbs = _readDouble(data['orbs'], _storedOrbs);
        final wins = _readInt(
          data['weeklyOperationWins'],
          _weeklyOperationWins,
        );

        transaction.set(docRef, {
          'weeklyCodeEventId': eventId,
          'weeklyCodeName': operation.title,
          'weeklyCodeCompletedAt': FieldValue.serverTimestamp(),
          'weeklyOperationWins': wins + 1,
          'points': currentPoints + 100,
          'orbs': currentOrbs + 0.1000,
          'updatedAt': FieldValue.serverTimestamp(),

      });

        _completedWeeklyEventId = eventId;
        _weeklyCodeCompleted = true;
        _weeklyOperationWins += 1;
        _points += 100;
        _storedOrbs += 0.1000;
        _displayOrbs += 0.1000;

  Widget _cryptoMovingTicker() {
    if (_cryptoQuotes.isEmpty) return const SizedBox.shrink();

    const itemWidth = 154.0;
    final travel = itemWidth * _cryptoQuotes.length;

    Widget group() {
      return Row(
        mainAxisSize: MainAxisSize.min,
        children:
            _cryptoQuotes.map((quote) {
              final positive = quote.change24h >= 0;
              final changeColor =
                  positive ? green : const Color(0xFFFF5C7A);
              return SizedBox(
                width: itemWidth,
                child: Padding(
                  padding: const EdgeInsets.only(right: 8),
                  child: Row(
                    children: [
                      Text(
                        quote.symbol,
                        style: const TextStyle(
                          color: Colors.white,
                          fontSize: 11,
                          fontWeight: FontWeight.w900,
                        ),
                      ),
                      const SizedBox(width: 6),
                      Expanded(
                        child: Text(
                          _formatUsd(quote.price),
                          maxLines: 1,
                          overflow: TextOverflow.fade,
                          softWrap: false,
                          style: const TextStyle(
                            color: Colors.white70,
                            fontSize: 11,
                            fontWeight: FontWeight.w700,
                          ),
                        ),
                      ),
                      const SizedBox(width: 4),
                      Text(
                        '${positive ? '+' : ''}${quote.change24h.toStringAsFixed(2)}%',
                        style: TextStyle(
                          color: changeColor,
                          fontSize: 10,
                          fontWeight: FontWeight.w900,
                        ),
                      ),
                    ],
                  ),
                ),
              );
            }).toList(),
      );
    }

    return Container(
      height: 35,
      clipBehavior: Clip.antiAlias,
      decoration: BoxDecoration(
        color: Colors.black.withOpacity(0.20),
        borderRadius: BorderRadius.circular(10),
        border: Border.all(color: Colors.white10),
      ),
      child: AnimatedBuilder(
        animation: _starsController,
        builder: (context, child) {
          final dx = -(_starsController.value * travel);
          return OverflowBox(
            alignment: Alignment.centerLeft,
            minWidth: 0,
            maxWidth: double.infinity,
            child: Transform.translate(
              offset: Offset(dx, 0),
              child: Row(
                mainAxisSize: MainAxisSize.min,
                children: [group(), group(), group()],
              ),
            ),
          );
        },
      ),
    );
  }

  Widget _newsPanel() {
    return InkWell(
      borderRadius: BorderRadius.circular(20),
      onTap: _openNewsPage,
      child: _glassCard(
        padding: const EdgeInsets.all(13),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              children: [
                const Text('ORQUBE NEWS', style: TextStyle(color: violet, fontSize: 15, fontWeight: FontWeight.w900)),
                const Spacer(),
                Text(_tr('더보기 >', 'View more >'), style: const TextStyle(color: Colors.white54, fontSize: 12)),
              ],
            ),
            const SizedBox(height: 10),
            StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
              stream: FirebaseFirestore.instance
                  .collection('orqube_news')
                  .where('isActive', isEqualTo: true)
                  .limit(50)
                  .snapshots(),
              builder: (context, snapshot) {
                final docs = snapshot.data?.docs.toList() ?? <QueryDocumentSnapshot<Map<String, dynamic>>>[];
                docs.sort((a, b) {
                  final at = a.data()['publishedAt'];
                  final bt = b.data()['publishedAt'];
                  final ad = at is Timestamp ? at.toDate() : DateTime.fromMillisecondsSinceEpoch(0);
                  final bd = bt is Timestamp ? bt.toDate() : DateTime.fromMillisecondsSinceEpoch(0);
                  return bd.compareTo(ad);
                });
                Map<String, dynamic>? latest(String type) {
                  for (final doc in docs) {
                    final data = doc.data();
                    final rawType = '${data['type'] ?? ''}'.toLowerCase();
                    final category = '${data['category'] ?? ''}'.toLowerCase();
                    if (rawType == type ||
                        (type == 'notice' && (category.contains('공지') || category == 'notice')) ||
                        (type == 'event' && (category.contains('이벤트') || category == 'event')) ||
                        (type == 'airsoft' && category.contains('airsoft'))) {
                      return data;
                    }
                  }
                  return null;
                }
                Widget tile(String type, String tag, String fallback) {
                  final data = latest(type);
                  return _NewsTile(
                    tag: tag,
                    title: data == null ? fallback : '${data['title'] ?? fallback}',
                    date: data == null ? 'NEW' : _newsDateLabel(data['publishedAt']),
                  );
                }
                return Column(
                  children: [
                    tile('notice', _tr('공지', 'NOTICE'), _tr('등록된 공지사항이 없습니다', 'No notice posted yet')),
                    tile('event', _tr('이벤트', 'EVENT'), _tr('등록된 이벤트가 없습니다', 'No event posted yet')),
                    tile('airsoft', 'AIRSOFT', _tr('등록된 AIRSOFT 소식이 없습니다', 'No AIRSOFT update yet')),
                  ],
                );
              },
            ),
          ],
        ),
      ),
    );
  }


  String _newsDateLabel(dynamic value) {
    if (value is! Timestamp) return 'NEW';
    final d = value.toDate().toLocal();
    return '${d.month.toString().padLeft(2, '0')}.${d.day.toString().padLeft(2, '0')}';
  }

  Future<void> _openProtectedPage(
    Widget page, {
    bool closeDrawer = false,
  }) async {
    // 기능 카드에서 진입할 때 현재 채굴 대시보드 route를 pop하지 않습니다.
    // 그래야 친구초대 화면의 뒤로가기가 채굴화면으로 정확히 돌아옵니다.
    if (closeDrawer) {
      Navigator.of(context).pop();
    }
    if (FirebaseAuth.instance.currentUser != null) {
      await Navigator.of(context).push(MaterialPageRoute(builder: (_) => page));
      return;
    }
    await Navigator.of(context).push(
      MaterialPageRoute(
        builder: (_) => LoginPage(
          onLoginSuccess: () async {
            if (!context.mounted) return;
            await Navigator.of(context).pushReplacement(
              MaterialPageRoute<void>(builder: (_) => page),
            );
          },
        ),
      ),
    );
  }

class _CryptoQuote {
  final String id;
  final String symbol;
  final String name;
  final double price;
  final double change24h;

  const _CryptoQuote({
    required this.id,
    required this.symbol,
    required this.name,
    required this.price,
    required this.change24h,
  });

  factory _CryptoQuote.fromJson(Map<String, dynamic> json) {
    double number(dynamic value) {
      if (value is num) return value.toDouble();
      return double.tryParse('${value ?? ''}') ?? 0;
    }

    return _CryptoQuote(
      id: '${json['id'] ?? ''}',
      symbol: '${json['symbol'] ?? ''}'.toUpperCase(),
      name: '${json['name'] ?? ''}',
      price: number(json['current_price']),
      change24h: number(
        json['price_change_percentage_24h_in_currency'] ??
            json['price_change_percentage_24h'],
      ),
    );
  }
}

class _SpacePainter extends CustomPainter {
  final double progress;

  const _SpacePainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final starPaint = Paint()..color = Colors.white.withOpacity(0.55);
    final purplePaint =
        Paint()..color = _MobileDashboardPageState.violet.withOpacity(0.16);

    for (int i = 0; i < 95; i++) {
      final x = (i * 47.0) % size.width;
      final y = (i * 83.0 + progress * 90) % size.height;
      // V22.15: retain the existing motion/count, reduce star size by 65%.
      final radius = (i % 3 + 1) * 0.38 * 0.35;
      canvas.drawCircle(Offset(x, y), radius, starPaint);
    }

    for (int i = 0; i < 8; i++) {
      final x = (i * 131.0 + progress * 50) % size.width;
      final y = (i * 97.0) % size.height;
      canvas.drawCircle(Offset(x, y), 42 + (i % 3) * 17, purplePaint);
    }
  }

  @override
  bool shouldRepaint(covariant _SpacePainter oldDelegate) =>
      oldDelegate.progress != progress;
}

class _MiningRingPainter extends CustomPainter {
  final double progress;
  final double pulse;
  final bool active;

  const _MiningRingPainter({
    required this.progress,
    required this.pulse,
    required this.active,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final center = size.center(Offset.zero);
    final radius = size.shortestSide / 2 - 14;
    final rect = Rect.fromCircle(center: center, radius: radius);

    final inactivePaint =
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 8
          ..color = Colors.white.withOpacity(0.08);

    canvas.drawCircle(center, radius, inactivePaint);

    final glowPaint =
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 16 + pulse * 4
          ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 10)
          ..shader = SweepGradient(
            transform: GradientRotation(progress * math.pi * 2),
            colors:
                active
                    ? const [
                      Color(0xFF8B2CF6),
                      Color(0xFFCF78FF),
                      Color(0xFFD7FF35),
                      Color(0xFF8B2CF6),
                    ]
                    : const [
                      Color(0xFF343445),
                      Color(0xFF646474),
                      Color(0xFF343445),
                    ],
          ).createShader(rect);

    canvas.drawArc(rect, 0, math.pi * 2, false, glowPaint);

    final ringPaint =
        Paint()
          ..style = PaintingStyle.stroke
          ..strokeWidth = 6
          ..strokeCap = StrokeCap.round
          ..shader = SweepGradient(
            transform: GradientRotation(progress * math.pi * 2),
            colors:
                active
                    ? const [
                      Color(0xFF8B2CF6),
                      Color(0xFFCF78FF),
                      Color(0xFFD7FF35),
                      Color(0xFF8B2CF6),
                    ]
                    : const [
                      Color(0xFF454554),
                      Color(0xFF777784),
                      Color(0xFF454554),
                    ],
          ).createShader(rect);

    canvas.drawArc(rect, 0, math.pi * 2, false, ringPaint);

    if (active) {
      final particlePaint = Paint()..color = _MobileDashboardPageState.lime;

      for (int i = 0; i < 16; i++) {
        final angle = progress * math.pi * 2 + i * math.pi / 8;
        final particleRadius = radius + 10 + (i % 3) * 3;
        final point = Offset(
          center.dx + math.cos(angle) * particleRadius,
          center.dy + math.sin(angle) * particleRadius,
        );
        canvas.drawCircle(point, i % 4 == 0 ? 2.2 : 1.1, particlePaint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant _MiningRingPainter oldDelegate) {
    return oldDelegate.progress != progress ||
        oldDelegate.pulse != pulse ||
        oldDelegate.active != active;
  }
}
